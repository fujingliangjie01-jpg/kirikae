<script>
const MODEL_KEY = "kirikae_model_confirmed_v2";
const CODE_KEY = "kirikae_code_confirmed_v1";
const MODEL_PATTERN = /[A-Z][0-9]{3}[A-Z]{1,3}/g;

const params = new URLSearchParams(location.search);
const mode = params.get("mode") === "code" ? "code" : "model";

const video = document.getElementById("video");
const canvas = document.getElementById("canvas");
const frame = document.getElementById("frame");
const guide = document.getElementById("guide");
const title = document.getElementById("title");
const backBtn = document.getElementById("backBtn");
const statusText = document.getElementById("statusText");
const candidate = document.getElementById("candidate");
const complete = document.getElementById("complete");
const completeValue = document.getElementById("completeValue");

let stream = null;
let scanning = true;
let busy = false;

let previousValue = "";
let sameCount = 0;

let barcodeDetector = null;
let ocrWorker = null;
let ocrReadyPromise = null;

if(mode === "code"){
  title.textContent = "コード確認";
  guide.textContent = "QRコードを枠内に入れてください";
  frame.classList.add("code");
}else{
  title.textContent = "機種読み込み";
  guide.textContent = "青枠に機種名を入れてください";
}

backBtn.addEventListener("click",()=>{
  stopCamera();
  location.href = "index.html";
});

function normalizeText(text){
  return String(text)
    .toUpperCase()
    .replace(/\s+/g,"")
    .replace(/[^A-Z0-9]/g,"");
}

function extractModel(text){
  const normalized = normalizeText(text);
  const matches = normalized.match(MODEL_PATTERN);
  if(!matches || !matches.length) return null;
  matches.sort((a,b)=>b.length-a.length);
  return matches[0];
}

function prepareOCR(){
  if(ocrReadyPromise) return ocrReadyPromise;

  ocrReadyPromise = (async()=>{
    ocrWorker = await Tesseract.createWorker("eng");

    await ocrWorker.setParameters({
      tessedit_char_whitelist:"ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789",
      tessedit_pageseg_mode:"7",
      preserve_interword_spaces:"0"
    });

    return ocrWorker;
  })();

  return ocrReadyPromise;
}

async function prepareBarcode(){
  if(!("BarcodeDetector" in window)) return;

  try{
    const supported = await BarcodeDetector.getSupportedFormats();

    const preferred = mode === "code"
      ? ["qr_code","data_matrix","code_128","code_39","code_93"]
      : ["code_128","code_39","code_93","qr_code","data_matrix"];

    const formats = preferred.filter(f=>supported.includes(f));

    barcodeDetector = new BarcodeDetector(
      formats.length ? {formats} : undefined
    );
  }catch(e){
    barcodeDetector = null;
  }
}

async function start(){
  try{
    await prepareBarcode();

    stream = await navigator.mediaDevices.getUserMedia({
      video:{
        facingMode:{ideal:"environment"},
        width:{ideal:1920},
        height:{ideal:1080}
      },
      audio:false
    });

    video.srcObject = stream;
    await video.play();

    if(mode === "model"){
      prepareOCR().catch(console.error);
    }

    statusText.textContent = "🔍 読み取り中…";
    setTimeout(loop,250);

  }catch(e){
    console.error(e);
    statusText.textContent = "カメラを起動できません";
  }
}

async function loop(){
  if(!scanning || busy) return;

  if(video.readyState < 2){
    setTimeout(loop,250);
    return;
  }

  busy = true;

  try{
    if(mode === "code"){
      await readCode();
    }else{
      const byBarcode = await readModelBarcode();
      if(!byBarcode && scanning){
        await readModelOCR();
      }
    }
  }catch(e){
    console.error(e);
  }finally{
    busy = false;
    if(scanning){
      setTimeout(loop,250);
    }
  }
}

async function readCode(){
  if(!barcodeDetector){
    statusText.textContent = "この端末ではQR読取方式を調整する必要があります";
    return;
  }

  try{
    const results = await barcodeDetector.detect(video);

    for(const result of results){
      const value = String(result.rawValue || "").trim();
      if(!value) continue;

      candidate.textContent = value;
      handleDetected(value,"QR/バーコード");
      return;
    }
  }catch(e){}
}

async function readModelBarcode(){
  if(!barcodeDetector) return false;

  try{
    const results = await barcodeDetector.detect(video);

    for(const result of results){
      const raw = String(result.rawValue || "");
      const model = extractModel(raw);

      if(model){
        candidate.textContent = model;
        handleDetected(model,"バーコード");
        return true;
      }
    }
  }catch(e){}

  return false;
}

async function readModelOCR(){
  const width = video.videoWidth;
  const height = video.videoHeight;

  if(!width || !height) return;

  const cropX = width * 0.10;
  const cropY = height * 0.56;
  const cropW = width * 0.80;
  const cropH = height * 0.19;

  const maxWidth = 1200;
  const scale = Math.min(1,maxWidth/cropW);

  const outW = Math.max(1,Math.round(cropW*scale));
  const outH = Math.max(1,Math.round(cropH*scale));

  canvas.width = outW;
  canvas.height = outH;

  const ctx = canvas.getContext("2d",{willReadFrequently:true});

  ctx.drawImage(
    video,
    cropX,cropY,cropW,cropH,
    0,0,outW,outH
  );

  try{
    const worker = await prepareOCR();
    const result = await worker.recognize(canvas);

    const normalized = normalizeText(result.data.text || "");
    const model = extractModel(normalized);

    if(model){
      candidate.textContent = model;
      handleDetected(model,"文字OCR");
    }
  }catch(e){
    console.error(e);
  }
}

function handleDetected(value,method){
  if(value === previousValue){
    sameCount++;
  }else{
    previousValue = value;
    sameCount = 1;
  }

  statusText.textContent = `確認 ${Math.min(sameCount,2)} / 2`;

  if(sameCount >= 2){
    confirm(value,method);
  }
}

function confirm(value,method){
  scanning = false;

  const now = new Date();

  const data = {
    value,
    method,
    timeISO:now.toISOString(),
    timeText:now.toLocaleString("ja-JP")
  };

  localStorage.setItem(
    mode === "code" ? CODE_KEY : MODEL_KEY,
    JSON.stringify(data)
  );

  completeValue.textContent = value;
  complete.classList.add("show");
  statusText.textContent = "✅ 読み込み完了";

  if(navigator.vibrate){
    navigator.vibrate(120);
  }

  setTimeout(()=>{
    stopCamera();
    location.href = "index.html";
  },1200);
}

function stopCamera(){
  scanning = false;

  if(stream){
    stream.getTracks().forEach(track=>track.stop());
    stream = null;
  }
}

start();
