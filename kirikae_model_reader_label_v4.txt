<script>
const STORAGE_KEY = "kirikae_model_confirmed_v2";

/*
  機種名ルール
  英字1文字 + 数字3文字 + 英字1〜3文字

  例:
  T629C
  T629AB
  T629ABC
*/
const MODEL_PATTERN = /[A-Z][0-9]{3}[A-Z]{1,3}/g;

const video = document.getElementById("camera");
const canvas = document.getElementById("canvas");
const cameraBtn = document.getElementById("cameraBtn");
const stopBtn = document.getElementById("stopBtn");
const retryBtn = document.getElementById("retryBtn");

const scanFrame = document.getElementById("scanFrame");
const scanLabel = document.getElementById("scanLabel");
const completeOverlay = document.getElementById("completeOverlay");
const overlayValue = document.getElementById("overlayValue");

const statusEl = document.getElementById("status");
const progressEl = document.getElementById("progress");
const currentTextEl = document.getElementById("currentText");
const candidateEl = document.getElementById("candidate");
const methodEl = document.getElementById("method");
const confirmedTextEl = document.getElementById("confirmedText");
const confirmedMetaEl = document.getElementById("confirmedMeta");

let stream = null;
let scanning = false;
let scanBusy = false;

let previousCandidate = "";
let sameCount = 0;
let confirmedData = null;

let barcodeDetector = null;
let ocrWorker = null;
let ocrReadyPromise = null;

restoreSavedData();


/* =========================
   BarcodeDetector準備
========================= */
(async () => {
  if (!("BarcodeDetector" in window)) return;

  try {
    const preferred = [
      "code_128",
      "code_39",
      "code_93",
      "ean_13",
      "ean_8",
      "upc_a",
      "upc_e",
      "qr_code",
      "data_matrix"
    ];

    const supported = await BarcodeDetector.getSupportedFormats();
    const formats = preferred.filter(f => supported.includes(f));

    barcodeDetector = new BarcodeDetector(
      formats.length ? { formats } : undefined
    );
  } catch (e) {
    barcodeDetector = null;
  }
})();


/* =========================
   OCR Worker準備
========================= */
function prepareOCR() {
  if (ocrReadyPromise) return ocrReadyPromise;

  ocrReadyPromise = (async () => {
    ocrWorker = await Tesseract.createWorker("eng");

    await ocrWorker.setParameters({
      tessedit_char_whitelist: "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789",
      tessedit_pageseg_mode: "7",
      preserve_interword_spaces: "0"
    });

    return ocrWorker;
  })();

  return ocrReadyPromise;
}


/* =========================
   カメラ起動
========================= */
cameraBtn.addEventListener("click", async () => {
  try {
    statusEl.textContent = "カメラ準備中…";

    stream = await navigator.mediaDevices.getUserMedia({
      video: {
        facingMode: { ideal: "environment" },
        width: { ideal: 1920 },
        height: { ideal: 1080 }
      },
      audio: false
    });

    video.srcObject = stream;
    await video.play();

    video.style.display = "block";
    scanFrame.style.display = "block";
    scanLabel.style.display = "block";

    cameraBtn.style.display = "none";
    stopBtn.style.display = "block";
    retryBtn.style.display = "none";

    completeOverlay.classList.remove("show");

    resetScanState();

    scanning = true;

    prepareOCR().catch(console.error);

    setTimeout(scanLoop, 250);

  } catch (error) {
    console.error(error);
    statusEl.textContent = "カメラを起動できません";
  }
});


/* =========================
   カメラ停止
========================= */
stopBtn.addEventListener("click", stopCamera);

function stopCamera() {
  scanning = false;

  if (stream) {
    stream.getTracks().forEach(track => track.stop());
    stream = null;
  }

  video.srcObject = null;
  video.style.display = "none";

  scanFrame.style.display = "none";
  scanLabel.style.display = "none";
  completeOverlay.classList.remove("show");

  cameraBtn.style.display = "block";
  stopBtn.style.display = "none";
  retryBtn.style.display = "none";

  statusEl.textContent = "停止中";
}


/* =========================
   再読取
========================= */
retryBtn.addEventListener("click", () => {
  if (!stream) return;

  completeOverlay.classList.remove("show");
  retryBtn.style.display = "none";

  resetScanState();

  scanning = true;
  setTimeout(scanLoop, 200);
});


function resetScanState() {
  previousCandidate = "";
  sameCount = 0;

  currentTextEl.textContent = "読み取り待ち";
  candidateEl.textContent = "未検出";
  methodEl.textContent = "-";

  progressEl.textContent = "確認 0 / 2";
  statusEl.textContent = "🔍 読み取り中…";
}


/* =========================
   メイン読取ループ
========================= */
async function scanLoop() {
  if (!scanning || scanBusy) return;

  if (video.readyState < 2) {
    setTimeout(scanLoop, 250);
    return;
  }

  scanBusy = true;

  try {
    statusEl.textContent = "🔍 読み取り中…";

    const barcodeFound = await tryBarcode();

    if (!barcodeFound && scanning) {
      await tryOCR();
    }

  } catch (e) {
    console.error(e);
  } finally {
    scanBusy = false;

    if (scanning) {
      setTimeout(scanLoop, 250);
    }
  }
}


/* =========================
   バーコード
========================= */
async function tryBarcode() {
  if (!barcodeDetector) return false;

  try {
    const barcodes = await barcodeDetector.detect(video);

    for (const barcode of barcodes) {
      const raw = normalizeText(barcode.rawValue);
      if (!raw) continue;

      currentTextEl.textContent = raw;
      methodEl.textContent = "バーコード";

      const model = extractModelName(raw);

      if (model) {
        candidateEl.textContent = model;
        handleCandidate(model, "バーコード");
        return true;
      }
    }

    return false;

  } catch (e) {
    return false;
  }
}


/* =========================
   OCR
========================= */
async function tryOCR() {
  const width = video.videoWidth;
  const height = video.videoHeight;

  if (!width || !height) return;

  /*
    写真のようなラベルを想定。
    下側の機種名を狙う細い読取範囲。
  */
  const cropX = width * 0.12;
  const cropY = height * 0.59;
  const cropW = width * 0.76;
  const cropH = height * 0.18;

  /*
    OCRを軽くするため縮小。
    ただしドット文字を潰しすぎないよう
    以前より大きめに残す。
  */
  const maxWidth = 1200;
  const scale = Math.min(1, maxWidth / cropW);

  const outW = Math.max(1, Math.round(cropW * scale));
  const outH = Math.max(1, Math.round(cropH * scale));

  canvas.width = outW;
  canvas.height = outH;

  const ctx = canvas.getContext("2d", { willReadFrequently: true });

  /*
    白黒2値化はしない。
    ドット印字を壊しにくくするため
    元画像の階調を残す。
  */
  ctx.drawImage(
    video,
    cropX, cropY, cropW, cropH,
    0, 0, outW, outH
  );

  try {
    const worker = await prepareOCR();
    const result = await worker.recognize(canvas);

    const normalized = normalizeText(result.data.text || "");

    if (!normalized) return;

    currentTextEl.textContent = normalized;
    methodEl.textContent = "文字OCR";

    const model = extractModelName(normalized);

    if (model) {
      candidateEl.textContent = model;
      handleCandidate(model, "文字OCR");
    }

  } catch (e) {
    console.error("OCR error", e);
  }
}


/* =========================
   文字整形
========================= */
function normalizeText(text) {
  return String(text)
    .toUpperCase()
    .replace(/\s+/g, "")
    .replace(/[^A-Z0-9]/g, "");
}


/* =========================
   機種名抽出
========================= */
function extractModelName(text) {
  const normalized = normalizeText(text);
  const matches = normalized.match(MODEL_PATTERN);

  if (!matches || !matches.length) {
    return null;
  }

  /*
    複数候補が出た場合は
    一番長い候補を優先。
  */
  matches.sort((a, b) => b.length - a.length);

  return matches[0];
}


/* =========================
   2回一致
========================= */
function handleCandidate(value, method) {
  currentTextEl.textContent = value;
  candidateEl.textContent = value;
  methodEl.textContent = method;

  if (value === previousCandidate) {
    sameCount++;
  } else {
    previousCandidate = value;
    sameCount = 1;
  }

  progressEl.textContent =
    `確認 ${Math.min(sameCount, 2)} / 2`;

  statusEl.textContent = "機種名を確認中…";

  if (sameCount >= 2) {
    confirmData(value, method);
  }
}


/* =========================
   確定・保持
========================= */
function confirmData(value, method) {
  const now = new Date();

  confirmedData = {
    value,
    method,
    timeISO: now.toISOString(),
    timeText: now.toLocaleString("ja-JP")
  };

  localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify(confirmedData)
  );

  showConfirmedData(confirmedData);

  statusEl.textContent = "✅ 読み取り完了";
  progressEl.textContent = "確認 2 / 2";

  scanning = false;

  overlayValue.textContent = value;
  completeOverlay.classList.add("show");

  retryBtn.style.display = "block";

  if (navigator.vibrate) {
    navigator.vibrate(120);
  }
}


/* =========================
   保持データ表示
========================= */
function showConfirmedData(data) {
  confirmedTextEl.textContent = data.value;
  confirmedMetaEl.textContent =
    `${data.method} / ${data.timeText}`;
}


/* =========================
   保存データ復元
========================= */
function restoreSavedData() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return;

    const data = JSON.parse(raw);
    if (!data || !data.value) return;

    confirmedData = data;
    showConfirmedData(data);

  } catch (e) {
    console.error(e);
  }
}


/* =========================
   外部参照用
========================= */
window.getConfirmedData = function() {
  return confirmedData;
};
