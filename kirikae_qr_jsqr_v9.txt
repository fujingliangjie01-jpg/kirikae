<script>
const MODEL_KEY="kirikae_model_confirmed_v8";
const CODE_KEY="kirikae_code_confirmed_v8";
const MODEL_PATTERN=/[A-Z][0-9]{3}[A-Z]{1,3}/g;

const startScreen=document.getElementById("startScreen");
const cameraScreen=document.getElementById("cameraScreen");
const inspectionScreen=document.getElementById("inspectionScreen");
const startModelCamera=document.getElementById("startModelCamera");
const rescanModel=document.getElementById("rescanModel");
const scanCode=document.getElementById("scanCode");
const cancelCamera=document.getElementById("cancelCamera");
const retryCamera=document.getElementById("retryCamera");
const modelValue=document.getElementById("modelValue");
const codeItem=document.getElementById("codeItem");
const codeStatus=document.getElementById("codeStatus");
const video=document.getElementById("video");
const canvas=document.getElementById("canvas");
const cameraTitle=document.getElementById("cameraTitle");
const scanFrame=document.getElementById("scanFrame");
const guide=document.getElementById("guide");
const completeOverlay=document.getElementById("completeOverlay");
const completeValue=document.getElementById("completeValue");
const cameraStatus=document.getElementById("cameraStatus");
const candidate=document.getElementById("candidate");
const helpText=document.getElementById("helpText");

let currentMode="model";
let stream=null;
let scanning=false;
let busy=false;
let cameraStarting=false;
let previousValue="";
let sameCount=0;
let barcodeDetector=null;
let ocrWorker=null;
let ocrReadyPromise=null;
let scanTimer=null;

function readJSON(key){
  try{
    const raw=localStorage.getItem(key);
    return raw?JSON.parse(raw):null;
  }catch(e){return null}
}

function showOnly(screen){
  startScreen.classList.add("hidden");
  cameraScreen.classList.add("hidden");
  inspectionScreen.classList.add("hidden");
  screen.classList.remove("hidden");
}

function showInspection(){
  stopCamera();
  const model=readJSON(MODEL_KEY);
  if(!model||!model.value){
    showOnly(startScreen);
    return;
  }
  modelValue.textContent=model.value;

  const code=readJSON(CODE_KEY);
  if(code&&code.value){
    codeStatus.textContent="✅ 完了";
    codeItem.classList.add("done");
  }else{
    codeStatus.textContent="未確認";
    codeItem.classList.remove("done");
  }
  showOnly(inspectionScreen);
}

(function init(){
  const model=readJSON(MODEL_KEY);
  if(model&&model.value) showInspection();
  else showOnly(startScreen);
})();

startModelCamera.addEventListener("click",()=>openCamera("model"));
rescanModel.addEventListener("click",()=>openCamera("model"));
scanCode.addEventListener("click",()=>openCamera("code"));
retryCamera.addEventListener("click",()=>openCamera(currentMode));

cancelCamera.addEventListener("click",()=>{
  stopCamera();
  if(readJSON(MODEL_KEY)) showInspection();
  else showOnly(startScreen);
});

async function openCamera(mode){
  /* 二重起動を完全防止 */
  if(cameraStarting) return;
  cameraStarting=true;

  stopCamera();
  currentMode=mode;
  previousValue="";
  sameCount=0;
  candidate.textContent="-";
  helpText.textContent="";
  retryCamera.classList.add("hidden");
  completeOverlay.classList.remove("show");

  if(mode==="code"){
    cameraTitle.textContent="コード確認";
    guide.textContent="QRコード全体を枠内に入れてください";
    scanFrame.classList.add("code");
  }else{
    cameraTitle.textContent="機種読み込み";
    guide.textContent="青枠に機種名を入れてください";
    scanFrame.classList.remove("code");
  }

  showOnly(cameraScreen);
  cameraStatus.classList.remove("error");
  cameraStatus.textContent="カメラ起動中…";

  try{
    if(!window.isSecureContext){
      throw new Error("NOT_SECURE");
    }

    if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
      throw new Error("NO_GETUSERMEDIA");
    }

    /* まず最も互換性の高い設定で1回だけ起動 */
    stream=await navigator.mediaDevices.getUserMedia({
      video:{facingMode:"environment"},
      audio:false
    });

    video.srcObject=stream;
    await video.play();

    await prepareBarcode();

    if(mode==="model"){
      prepareOCR().catch(()=>{});
    }

    scanning=true;
    cameraStatus.textContent="🔍 読み取り中…";
    scanTimer=setTimeout(scanLoop,300);

  }catch(e){
    console.error(e);
    stopCamera();
    cameraStatus.classList.add("error");

    if(e && e.name==="NotAllowedError"){
      cameraStatus.textContent="カメラの使用が許可されていません";
      helpText.textContent="Safariのサイト設定で「カメラ」を許可してから「カメラを再起動」を押してください。";
    }else if(e && e.name==="NotFoundError"){
      cameraStatus.textContent="使用できるカメラが見つかりません";
    }else if(String(e && e.message)==="NOT_SECURE"){
      cameraStatus.textContent="HTTPSではないためカメラを起動できません";
    }else if(String(e && e.message)==="NO_GETUSERMEDIA"){
      cameraStatus.textContent="このブラウザではカメラ起動に対応していません";
      helpText.textContent="Safari本体でGitHub Pagesを開いてください。";
    }else{
      cameraStatus.textContent="カメラを起動できません";
      helpText.textContent="Safari本体で開き直すか、カメラ権限を確認してください。";
    }

    retryCamera.classList.remove("hidden");
  }finally{
    cameraStarting=false;
  }
}

function stopCamera(){
  scanning=false;
  busy=false;

  if(scanTimer){
    clearTimeout(scanTimer);
    scanTimer=null;
  }

  if(stream){
    stream.getTracks().forEach(t=>t.stop());
    stream=null;
  }

  video.srcObject=null;
}

function prepareOCR(){
  if(ocrReadyPromise)return ocrReadyPromise;
  ocrReadyPromise=(async()=>{
    ocrWorker=await Tesseract.createWorker("eng");
    await ocrWorker.setParameters({
      tessedit_char_whitelist:"ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789",
      tessedit_pageseg_mode:"7",
      preserve_interword_spaces:"0"
    });
    return ocrWorker;
  })();
  return ocrReadyPromise;
}

async function prepareBarcode(){
  barcodeDetector=null;
  if(!("BarcodeDetector" in window))return;
  try{
    const supported=await BarcodeDetector.getSupportedFormats();
    const preferred=currentMode==="code"
      ?["qr_code","data_matrix","code_128","code_39","code_93"]
      :["code_128","code_39","code_93","qr_code","data_matrix"];
    const formats=preferred.filter(f=>supported.includes(f));
    barcodeDetector=new BarcodeDetector(formats.length?{formats}:undefined);
  }catch(e){
    barcodeDetector=null;
  }
}

function normalizeText(text){
  return String(text).toUpperCase().replace(/\s+/g,"").replace(/[^A-Z0-9]/g,"");
}

function extractModel(text){
  const matches=normalizeText(text).match(MODEL_PATTERN);
  if(!matches||!matches.length)return null;
  matches.sort((a,b)=>b.length-a.length);
  return matches[0];
}

async function scanLoop(){
  if(!scanning||busy)return;

  if(video.readyState<2){
    scanTimer=setTimeout(scanLoop,300);
    return;
  }

  busy=true;
  try{
    if(currentMode==="code"){
      await readCode();
    }else{
      const hit=await readModelBarcode();
      if(!hit&&scanning)await readModelOCR();
    }
  }catch(e){
    console.error(e);
  }finally{
    busy=false;
    if(scanning)scanTimer=setTimeout(scanLoop,currentMode==="code"?120:300);
  }
}

async function readCode(){
  /*
    iPhone対策：
    BarcodeDetectorに依存せず、まずjsQRで映像フレームを直接解析。
    同じQR内容を2回読めた時だけ確定する。
  */
  const w=video.videoWidth;
  const h=video.videoHeight;

  if(!w||!h)return;

  const maxW=900;
  const scale=Math.min(1,maxW/w);
  const outW=Math.max(1,Math.round(w*scale));
  const outH=Math.max(1,Math.round(h*scale));

  canvas.width=outW;
  canvas.height=outH;

  const ctx=canvas.getContext("2d",{willReadFrequently:true});
  ctx.drawImage(video,0,0,w,h,0,0,outW,outH);

  try{
    if(typeof jsQR==="function"){
      const image=ctx.getImageData(0,0,outW,outH);
      const qr=jsQR(image.data,outW,outH,{
        inversionAttempts:"attemptBoth"
      });

      if(qr && qr.data){
        const value=String(qr.data).trim();

        if(value){
          candidate.textContent=value;
          handleDetected(value,"QR");
          return;
        }
      }
    }
  }catch(e){
    console.error("jsQR",e);
  }

  /*
    jsQRで見つからなかった場合だけ
    BarcodeDetectorも使える端末では補助的に確認。
  */
  if(barcodeDetector){
    try{
      const results=await barcodeDetector.detect(video);

      for(const r of results){
        const value=String(r.rawValue||"").trim();

        if(!value)continue;

        candidate.textContent=value;
        handleDetected(value,"QR/バーコード");
        return;
      }
    }catch(e){}
  }

  cameraStatus.textContent="🔍 QRコードを読み取り中…";
}
async function readModelBarcode(){
  if(!barcodeDetector)return false;
  try{
    const results=await barcodeDetector.detect(video);
    for(const r of results){
      const model=extractModel(r.rawValue||"");
      if(model){
        candidate.textContent=model;
        handleDetected(model,"バーコード");
        return true;
      }
    }
  }catch(e){}
  return false;
}

async function readModelOCR(){
  const w=video.videoWidth,h=video.videoHeight;
  if(!w||!h)return;

  const x=w*.10,y=h*.56,cw=w*.80,ch=h*.19;
  const scale=Math.min(1,1200/cw);

  canvas.width=Math.max(1,Math.round(cw*scale));
  canvas.height=Math.max(1,Math.round(ch*scale));

  const ctx=canvas.getContext("2d",{willReadFrequently:true});
  ctx.drawImage(video,x,y,cw,ch,0,0,canvas.width,canvas.height);

  try{
    const worker=await prepareOCR();
    const result=await worker.recognize(canvas);
    const model=extractModel(result.data.text||"");

    if(model){
      candidate.textContent=model;
      handleDetected(model,"文字OCR");
    }
  }catch(e){}
}

function handleDetected(value,method){
  if(value===previousValue)sameCount++;
  else{
    previousValue=value;
    sameCount=1;
  }

  cameraStatus.textContent=`確認 ${Math.min(sameCount,2)} / 2`;

  if(sameCount>=2){
    confirmRead(value,method);
  }
}

function confirmRead(value,method){
  scanning=false;
  if(scanTimer){
    clearTimeout(scanTimer);
    scanTimer=null;
  }

  const now=new Date();

  localStorage.setItem(
    currentMode==="code"?CODE_KEY:MODEL_KEY,
    JSON.stringify({
      value,
      method,
      timeISO:now.toISOString(),
      timeText:now.toLocaleString("ja-JP")
    })
  );

  completeValue.textContent=value;
  completeOverlay.classList.add("show");
  cameraStatus.textContent="✅ 読み込み完了";

  if(navigator.vibrate)navigator.vibrate(120);

  setTimeout(showInspection,1300);
}
