<script>
const MODEL_KEY = "kirikae_model_confirmed_v2";
const CODE_KEY = "kirikae_code_confirmed_v1";

const modelValue = document.getElementById("modelValue");
const inspectionArea = document.getElementById("inspectionArea");
const scanModelBtn = document.getElementById("scanModelBtn");
const scanCodeBtn = document.getElementById("scanCodeBtn");
const codeItem = document.getElementById("codeItem");
const codeStatus = document.getElementById("codeStatus");
const codeValue = document.getElementById("codeValue");

function readJSON(key){
  try{
    const raw = localStorage.getItem(key);
    return raw ? JSON.parse(raw) : null;
  }catch(e){
    return null;
  }
}

function render(){
  const model = readJSON(MODEL_KEY);

  if(model && model.value){
    modelValue.textContent = model.value;
    modelValue.className = "model";
    inspectionArea.classList.remove("hidden");
    scanModelBtn.textContent = "📷 機種を読み直す";
  }else{
    modelValue.textContent = "機種情報なし";
    modelValue.className = "empty";
    inspectionArea.classList.add("hidden");
    scanModelBtn.textContent = "📷 機種を読み取る";
  }

  const code = readJSON(CODE_KEY);

  if(code && code.value){
    codeValue.textContent = code.value;
    codeValue.className = "model";
    codeStatus.textContent = "✅ 完了";
    codeItem.classList.add("done");
    scanCodeBtn.textContent = "再読取";
  }else{
    codeValue.textContent = "未読取";
    codeValue.className = "empty";
    codeStatus.textContent = "未確認";
    codeItem.classList.remove("done");
    scanCodeBtn.textContent = "📷 QR確認";
  }
}

scanModelBtn.addEventListener("click",()=>{
  location.href = "camera.html?mode=model";
});

scanCodeBtn.addEventListener("click",()=>{
  location.href = "camera.html?mode=code";
});

window.addEventListener("pageshow",render);
render();
</script>
